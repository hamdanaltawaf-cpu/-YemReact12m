import { Flame, Sparkles } from "lucide-react";
import { type LibrarySort } from "@/lib/librarySorting";

export type { LibrarySort } from "@/lib/librarySorting";

export function LibraryHeader({ hasQuery, sort, onSortChange }: { hasQuery: boolean; sort: LibrarySort; onSortChange: (value: LibrarySort) => void }) {
  const options = [
    ...(hasQuery ? [{ value: "relevance" as const, label: "الأكثر صلة", Icon: Sparkles }] : []),
    { value: "latest" as const, label: "الأحدث", Icon: Sparkles },
    { value: "popular" as const, label: "الأكثر رواجًا", Icon: Flame },
  ];
  return <header className="library-header" aria-label={hasQuery ? "أدوات نتائج البحث" : "أدوات تصفح المكتبة"}><div className="library-header__controls library-header__controls--spread"><span className="library-header__title font-heading">مكتبة لا تُحْصَى</span><div className="library-sort" role="group" aria-label="ترتيب الرياكشنات">{options.map(({ value, label, Icon }) => <button key={value} type="button" className={`library-sort__button ${sort === value ? "is-active" : ""}`} aria-pressed={sort === value} onClick={() => onSortChange(value)}><Icon size={13} aria-hidden="true" /><span>{label}</span></button>)}</div></div></header>;
}

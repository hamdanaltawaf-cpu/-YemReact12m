import { Toaster } from "@/components/ui/sonner";
import { TooltipProvider } from "@/components/ui/tooltip";
import { lazy, Suspense } from "react";
import { Route, Switch } from "wouter";
import ErrorBoundary from "./components/ErrorBoundary";
import { SiteSidebar } from "./components/SiteSidebar";
import { ThemeProvider } from "./contexts/ThemeContext";
import Home from "./pages/Home";
import NotFound from "./pages/NotFound";

const Account = lazy(() => import("./pages/Account"));
const Admin = lazy(() => import("./pages/Admin"));
const AdminOperations = lazy(() => import("./pages/AdminOperations"));
const CategoryArchive = lazy(() => import("./pages/CategoryArchive"));
const CollectionArchive = lazy(() => import("./pages/CollectionArchive"));
const Groups = lazy(() => import("./pages/Groups"));
const ReactionDetail = lazy(() => import("./pages/ReactionDetail"));
const SubmitReaction = lazy(() => import("./pages/SubmitReaction"));

function Router() {
  return <Suspense fallback={<div className="route-loading" role="status" aria-live="polite">جارٍ فتح الصفحة…</div>}><Switch>{import.meta.env.DEV && <Route path="/__visual-admin" component={Admin} />}<Route path="/" component={Home} /><Route path="/groups" component={Groups} /><Route path="/r/:publicId" component={ReactionDetail} /><Route path="/category/:categoryId" component={CategoryArchive} /><Route path="/categories/:categoryId" component={CategoryArchive} /><Route path="/collection/:slug" component={CollectionArchive} /><Route path="/collections/:slug" component={CollectionArchive} /><Route path="/account" component={Account} /><Route path="/submit" component={SubmitReaction} /><Route path="/admin/operations" component={AdminOperations} /><Route path="/admin" component={Admin} /><Route component={NotFound} /></Switch></Suspense>;
}

export default function App() { return <ErrorBoundary><ThemeProvider defaultTheme="dark"><TooltipProvider><Toaster /><SiteSidebar /><main className="site-main-frame"><Router /></main></TooltipProvider></ThemeProvider></ErrorBoundary>; }
